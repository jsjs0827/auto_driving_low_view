cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(r36_camera_compare LANGUAGES CXX)
  include(CTest)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Live camera comparison requires Linux camera SDK and process measurements")
endif()
get_filename_component(_camera_product_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET perception_native_config)
  add_subdirectory("${_camera_product_root}/scheduling/config" "${CMAKE_CURRENT_BINARY_DIR}/native_config")
endif()
add_library(camera_compare_control STATIC src/control.cpp src/report.cpp)
target_compile_features(camera_compare_control PUBLIC cxx_std_17)
target_include_directories(camera_compare_control PUBLIC include)
target_link_libraries(camera_compare_control PUBLIC perception_native_config)
target_compile_options(camera_compare_control PRIVATE -Wall -Wextra -Wpedantic)
add_executable(r36_camera_compare src/main.cpp)
target_link_libraries(r36_camera_compare PRIVATE camera_compare_control)
install(TARGETS r36_camera_compare RUNTIME DESTINATION lib/perception_ros_so_demo)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(camera_compare_contract_test tests/contracts.cpp)
  target_link_libraries(camera_compare_contract_test PRIVATE camera_compare_control)
  add_test(NAME camera_compare_contract COMMAND camera_compare_contract_test
    "${_camera_product_root}" "${CMAKE_CURRENT_BINARY_DIR}/contract-output")
endif()
