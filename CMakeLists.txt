cmake_minimum_required(VERSION 3.20)
project(perception_stereo_cpp_v8_dag VERSION 8.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/utf-8 /EHsc)
endif()
include(CTest)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/production_input.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/algorithms.cmake")

option(PERCEPTION_BUILD_SDK "Build the SDK from this source tree" ON)
option(PERCEPTION_BUILD_V0_HOST_CORE "Build the source-neutral upstream entry core (no SHM driver)" OFF)
option(PERCEPTION_BUILD_CONFIG_TOOL "Build the native JSON configuration and launcher" ON)
option(PERCEPTION_BUILD_COMPARE "Build native comparison preparation and reports" ON)
option(PERCEPTION_BUILD_COMPARE_RUNNER "Build the SDK comparison worker" ON)
option(PERCEPTION_BUILD_CAPTURE "Build the camera triplet recorder (Linux)" ON)
option(PERCEPTION_BUILD_CAMERA_BENCHMARK "Build native direct-camera experiment controller" ON)
option(PERCEPTION_BUILD_WINDOWS_DEBUG "Build native Windows offline debugging entry" ${WIN32})
option(PERCEPTION_BUILD_SLIP_RUNNER "Build camera/recording motion and slip runner" ON)
option(PERCEPTION_BUILD_RVIZ "Build JSON-configured ADB/RViz display support" ON)
option(R33_WITH_RECHARGE "Build recharge backend with its optional dependencies" OFF)
option(R36_COMPARE_WITH_HORIZON "Enable Horizon stereo in the comparison worker" OFF)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/install/perception_ros_so_demo" CACHE PATH "Install prefix" FORCE)
endif()

add_subdirectory(features/visual_motion)
if(PERCEPTION_BUILD_SDK)
  add_subdirectory(sdk)
  if(TARGET perception_sdk AND NOT TARGET perception::sdk)
    add_library(perception::sdk ALIAS perception_sdk)
  endif()
endif()
if(PERCEPTION_BUILD_V0_HOST_CORE)
  if(NOT TARGET perception::sdk)
    find_package(perception_full_shared_sdk CONFIG REQUIRED)
  endif()
  if(NOT TARGET sdk_bridge)
    add_subdirectory(features/sdk_bridge)
  endif()
  add_subdirectory(features/v0_host)
endif()
if(PERCEPTION_BUILD_CONFIG_TOOL)
  add_subdirectory(scheduling/config)
endif()
if(PERCEPTION_BUILD_COMPARE)
  set(R36_COMPARE_INSTALL_DESTINATION lib/perception_ros_so_demo)
  add_subdirectory(compare/native)
endif()
if(PERCEPTION_BUILD_COMPARE_RUNNER)
  add_subdirectory(compare/runner)
endif()
if(PERCEPTION_BUILD_CAPTURE)
  add_subdirectory(compare/capture)
endif()
if(PERCEPTION_BUILD_CAMERA_BENCHMARK)
  if(NOT TARGET perception_native_config)
    message(FATAL_ERROR "Camera benchmarks require PERCEPTION_BUILD_CONFIG_TOOL=ON")
  endif()
  add_subdirectory(compare/camera)
endif()
if(PERCEPTION_BUILD_WINDOWS_DEBUG)
  if(NOT WIN32)
    message(FATAL_ERROR "PERCEPTION_BUILD_WINDOWS_DEBUG requires a Windows target")
  endif()
  add_subdirectory(windows)
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT ${PERCEPTION_WINDOWS_DEBUG_TARGET})
endif()
if(PERCEPTION_BUILD_SLIP_RUNNER)
  add_subdirectory(features/slip_app)
endif()
if(PERCEPTION_BUILD_RVIZ)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(R36_RVIZ_BUILD_PUBLISHER ON CACHE BOOL "Build native X5 RViz publisher")
  endif()
  set(R36_RVIZ_BUILD_SDK_OUTPUT ON CACHE BOOL "Build SDK RViz display adapter")
  add_subdirectory(features/rviz)
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/install_runtime_config.cmake")
perception_install_runtime_config("${CMAKE_CURRENT_SOURCE_DIR}" "share/perception_ros_so_demo")
install(DIRECTORY tools/ DESTINATION share/perception_ros_so_demo/tools
  USE_SOURCE_PERMISSIONS PATTERN "*.py" EXCLUDE PATTERN "*.pyc" EXCLUDE PATTERN "__pycache__" EXCLUDE)
