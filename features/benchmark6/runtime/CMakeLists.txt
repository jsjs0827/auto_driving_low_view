cmake_minimum_required(VERSION 3.20)
project(perception_benchmark6_runtime VERSION 3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
option(AB_CONTRACT_ONLY "Build portable configuration/statistics checks only" OFF)
option(AB_WITH_X5 "Build genuine original BPU and Stereo implementations" OFF)
option(AB_WITH_ROS "Build ROS2 image subscriber and source publisher" OFF)
# Use the same reviewed backend as the ordinary SDK and full DAG. The embedded
# vendor/perception tree is a historical snapshot, not the active V2 algorithm.
get_filename_component(AB_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT EXISTS "${AB_VENDOR}/sdk/CMakeLists.txt" OR
   NOT EXISTS "${AB_VENDOR}/features/pointcloud/backend/CMakeLists.txt")
  message(FATAL_ERROR "Benchmark6 requires the canonical V2 project; stale embedded mapping is not a fallback")
endif()
message(STATUS "Benchmark6 canonical backend: ${AB_VENDOR}")
add_library(ab_contract INTERFACE)
target_include_directories(ab_contract INTERFACE include "${AB_VENDOR}/scheduling/config/include")
if(MSVC)
  target_compile_options(ab_contract INTERFACE /utf-8 /EHsc)
endif()
add_executable(ab_config_check src/config_check.cpp)
target_link_libraries(ab_config_check PRIVATE ab_contract)
if(BUILD_TESTING)
  add_executable(ab_contract_tests tests/contracts.cpp)
  target_link_libraries(ab_contract_tests PRIVATE ab_contract)
  add_test(NAME ab_config_geometry_statistics COMMAND ab_contract_tests)
endif()
if(AB_CONTRACT_ONLY)
  return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The real benchmark measures Linux process resources. Use AB_CONTRACT_ONLY on Windows.")
endif()
find_package(Threads REQUIRED)
set(R20_WITH_BPU ${AB_WITH_X5} CACHE BOOL "" FORCE)
set(R33_WITH_MAPPING ${AB_WITH_X5} CACHE BOOL "" FORCE)
set(R33_WITH_RECHARGE OFF CACHE BOOL "" FORCE)
set(R33_MAPPING_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(R29_WITH_ORT OFF CACHE BOOL "" FORCE)
if(AB_WITH_X5)
  set(_ab_testing ${BUILD_TESTING})
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${AB_VENDOR}/sdk" "${CMAKE_CURRENT_BINARY_DIR}/canonical_sdk")
  set(BUILD_TESTING ${_ab_testing} CACHE BOOL "" FORCE)
else()
  # The Ubuntu ROS bridge never loads models or executes mapping. Keep its
  # build independent of ARM SDK/PCL/ONNX dependencies.
  add_library(platform_metrics STATIC
    "${AB_VENDOR}/features/metrics/backend/src/platform_metrics.cpp"
    "${AB_VENDOR}/features/metrics/backend/src/bpu_metrics.cpp")
  target_include_directories(platform_metrics PUBLIC "${AB_VENDOR}/features/metrics/backend/include")
  target_link_libraries(platform_metrics PUBLIC Threads::Threads)
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
include("${AB_VENDOR}/cmake/nv12.cmake")
add_library(ab_runtime STATIC src/input.cpp src/tcp_input.cpp src/pipeline.cpp
  "${AB_VENDOR}/features/camera/src/mo_camera.cpp")
target_include_directories(ab_runtime PUBLIC include
  "${AB_VENDOR}/sdk/include" "${AB_VENDOR}/features/camera/include"
  "${AB_VENDOR}/features/camera/vendor/mo" "${AB_VENDOR}/core/include"
  "${AB_VENDOR}/features/pointcloud/backend/include"
  "${AB_VENDOR}/features/pointcloud/vendor/include"
  "${AB_VENDOR}/features/metrics/backend/include")
target_include_directories(ab_runtime SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ab_runtime PUBLIC ab_contract
  platform_metrics stereo_nv12 ${OpenCV_LIBS} ${CMAKE_DL_LIBS})
if(AB_WITH_X5)
  target_link_libraries(ab_runtime PUBLIC perception_sdk r33_mapping_core)
endif()
if(AB_WITH_X5)
  set(STEREO_BUILD_HORIZON_RUNNER ON CACHE BOOL "" FORCE)
  set(STEREO_BUILD_CODEC_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${AB_VENDOR}/features/stereo/runner" "${CMAKE_CURRENT_BINARY_DIR}/canonical_stereo_runner")
  add_library(ab_original_dnn STATIC vendor/original/horizon_dnn_runner.cpp src/original_adapter.cpp)
  target_include_directories(ab_original_dnn PRIVATE
    include "${CMAKE_CURRENT_SOURCE_DIR}/vendor/original/include"
    "${AB_VENDOR}/scheduling/config/include" "${AB_VENDOR}/features/pointcloud/backend/include"
    ${OpenCV_INCLUDE_DIRS} "${HORIZON_ROOT}/include")
  # Original and optimized runners have the same C++ names but different layouts.
  # Isolate only the original translation units, keeping the original source bytes intact.
  target_compile_definitions(ab_original_dnn PRIVATE PERCEP_DNN_ENABLE=1 ai_node_percep_demo=ab_original_dnn_namespace)
  target_compile_options(ab_original_dnn PRIVATE -O3 -fno-fast-math)
  target_link_libraries(ab_original_dnn PUBLIC "${HORIZON_DNN_LIBRARY}" Threads::Threads)
  target_compile_definitions(ab_runtime PUBLIC AB_WITH_X5=1)
  target_link_libraries(ab_runtime PUBLIC ab_original_dnn horizon_stereo)
endif()
if(AB_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  target_sources(ab_runtime PRIVATE src/ros_input.cpp)
  target_compile_definitions(ab_runtime PUBLIC AB_WITH_ROS=1)
  ament_target_dependencies(ab_runtime PUBLIC rclcpp sensor_msgs)
  add_executable(ab_result_replay src/result_replay.cpp)
  target_link_libraries(ab_result_replay PRIVATE ab_contract ${OpenCV_LIBS})
  target_include_directories(ab_result_replay SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
  ament_target_dependencies(ab_result_replay PUBLIC rclcpp sensor_msgs std_msgs)
  install(TARGETS ab_result_replay RUNTIME DESTINATION bin COMPONENT ABRuntime)
endif()
add_executable(ab_benchmark src/main.cpp)
target_link_libraries(ab_benchmark PRIVATE ab_runtime)
add_executable(ab_report src/report.cpp src/pairing.cpp)
target_link_libraries(ab_report PRIVATE ab_contract)
if(BUILD_TESTING)
  add_executable(ab_input_tests tests/input_tests.cpp)
  target_link_libraries(ab_input_tests PRIVATE ab_runtime)
  add_test(NAME ab_offline_and_nv12 COMMAND ab_input_tests)
  add_executable(ab_pairing_tests tests/pairing_tests.cpp src/pairing.cpp)
  target_link_libraries(ab_pairing_tests PRIVATE ab_contract)
  add_test(NAME ab_pairing_requires_same_pixels COMMAND ab_pairing_tests)
  if(NOT AB_WITH_X5)
    add_library(ab_camera_fixture SHARED "${AB_VENDOR}/validation/camera_fixture.cpp")
    target_include_directories(ab_camera_fixture PRIVATE "${AB_VENDOR}/features/camera/vendor/mo")
    add_executable(ab_camera_tests tests/camera_tests.cpp)
    target_link_libraries(ab_camera_tests PRIVATE ab_runtime ${CMAKE_DL_LIBS})
    add_dependencies(ab_camera_tests ab_camera_fixture)
    add_test(NAME ab_camera_ABI_ownership_and_calibration
      COMMAND ab_camera_tests $<TARGET_FILE:ab_camera_fixture>)
  endif()
endif()
install(TARGETS ab_benchmark ab_report ab_config_check RUNTIME DESTINATION bin COMPONENT ABRuntime)
if(TARGET perception_sdk)
  install(TARGETS perception_sdk LIBRARY DESTINATION lib COMPONENT ABRuntime)
endif()
install(DIRECTORY config models data DESTINATION . COMPONENT ABRuntime)
install(PROGRAMS tools/run_matrix.sh tools/run_one.sh tools/run_ros_adb.sh DESTINATION tools COMPONENT ABRuntime)
install(FILES README_zh.md DESTINATION . COMPONENT ABRuntime)
