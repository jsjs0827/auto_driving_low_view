cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(perception_benchmark6 LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory(../visual_motion "${CMAKE_CURRENT_BINARY_DIR}/motion")
  add_subdirectory(../../scheduling/config "${CMAKE_CURRENT_BINARY_DIR}/config")
endif()
add_library(benchmark6_core STATIC src/common.cpp src/quality.cpp src/plan_report.cpp)
target_include_directories(benchmark6_core PUBLIC include PRIVATE ../../third_party/stb)
target_link_libraries(benchmark6_core PUBLIC perception_native_config)
target_compile_features(benchmark6_core PUBLIC cxx_std_17)
add_executable(r36_benchmark6 src/main.cpp)
target_link_libraries(r36_benchmark6 PRIVATE benchmark6_core)
install(TARGETS r36_benchmark6 RUNTIME DESTINATION bin)
