cmake_minimum_required(VERSION 3.16)
project(r36_horizon_stereo VERSION 0.4.1 LANGUAGES CXX)
option(STEREO_BUILD_HORIZON_RUNNER "Build real Horizon model library and CLI" ON)
option(STEREO_BUILD_CODEC_TESTS "Build pure codec tests against actual Horizon headers" ON)
set(HORIZON_ROOT "" CACHE PATH "Matching Horizon runtime prefix (include and lib)")
if(NOT DEFINED STEREO_RUNNER_INSTALL_DIR)
  set(STEREO_RUNNER_INSTALL_DIR bin CACHE STRING "CLI install destination relative to prefix")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_path(HORIZON_INCLUDE_DIR dnn/hb_dnn.h HINTS "${HORIZON_ROOT}/include" REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/nv12.cmake")
add_library(stereo_tensor_codec STATIC src/tensor_codec.cpp)
target_compile_features(stereo_tensor_codec PUBLIC cxx_std_17)
target_include_directories(stereo_tensor_codec PUBLIC include)
# Horizon prefixes may also ship an older opencv2 tree. Prefer the headers
# belonging to find_package(OpenCV), including for targets consuming this codec.
target_include_directories(stereo_tensor_codec SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS} "${HORIZON_INCLUDE_DIR}")
target_link_libraries(stereo_tensor_codec PUBLIC stereo_nv12 ${OpenCV_LIBS})
if(STEREO_BUILD_HORIZON_RUNNER)
  find_package(Threads REQUIRED)
  find_library(HORIZON_DNN_LIBRARY NAMES dnn HINTS "${HORIZON_ROOT}/lib" "${HORIZON_ROOT}/lib/hbbpu" REQUIRED)
  add_library(horizon_stereo STATIC src/horizon_stereo.cpp)
  target_compile_features(horizon_stereo PUBLIC cxx_std_17)
  target_link_libraries(horizon_stereo PUBLIC stereo_tensor_codec PRIVATE "${HORIZON_DNN_LIBRARY}" Threads::Threads)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # libdnn's DT_NEEDED libraries live outside its lib directory on X5.
    # Propagate link-time searches to every executable using this static library;
    # rpath-link does not add a build-machine path to the runtime ELF.
    get_filename_component(_horizon_libdir "${HORIZON_DNN_LIBRARY}" DIRECTORY)
    get_filename_component(_horizon_prefix "${_horizon_libdir}/.." ABSOLUTE)
    set(_horizon_dependency_dirs ${PSDK_LINK_DEPENDENCY_DIRS}
      "${_horizon_libdir}" "${_horizon_libdir}/hbbpu"
      "${_horizon_libdir}/aarch64-linux-gnu" "${_horizon_prefix}/hobot/lib")
    if(HORIZON_ROOT)
      list(APPEND _horizon_dependency_dirs
        "${HORIZON_ROOT}/lib" "${HORIZON_ROOT}/lib/hbbpu"
        "${HORIZON_ROOT}/lib/aarch64-linux-gnu" "${HORIZON_ROOT}/hobot/lib")
    endif()
    list(REMOVE_DUPLICATES _horizon_dependency_dirs)
    foreach(directory IN LISTS _horizon_dependency_dirs)
      if(IS_DIRECTORY "${directory}")
        target_link_options(horizon_stereo INTERFACE "-Wl,-rpath-link,${directory}")
      endif()
    endforeach()
  endif()
  add_executable(stereo_model_runner src/main.cpp)
  target_include_directories(stereo_model_runner PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../compare/runner/include")
  target_link_libraries(stereo_model_runner PRIVATE horizon_stereo)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    get_filename_component(_horizon_libdir "${HORIZON_DNN_LIBRARY}" DIRECTORY)
    target_link_options(stereo_model_runner PRIVATE "-Wl,--no-undefined" "-Wl,-rpath-link,${_horizon_libdir}")
  endif()
  install(TARGETS stereo_model_runner RUNTIME DESTINATION "${STEREO_RUNNER_INSTALL_DIR}")
endif()
if(STEREO_BUILD_CODEC_TESTS)
  enable_testing()
  add_executable(stereo_tensor_codec_test tests/tensor_codec_test.cpp)
  target_link_libraries(stereo_tensor_codec_test PRIVATE stereo_tensor_codec)
  add_test(NAME stereo_tensor_codec COMMAND stereo_tensor_codec_test)
  add_executable(stereo_nv12_test tests/nv12_test.cpp)
  target_link_libraries(stereo_nv12_test PRIVATE stereo_tensor_codec)
  add_test(NAME stereo_direct_nv12 COMMAND stereo_nv12_test)
endif()
