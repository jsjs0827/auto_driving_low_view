if(R36_BUILD_ALGORITHM_DEMO AND R36_BUILD_STEREO_HELPER_TESTS)
  enable_testing()
  add_library(camera_lifecycle_fixture SHARED "${R36_PROJECT_ROOT}/validation/camera_fixture.cpp")
  target_compile_features(camera_lifecycle_fixture PRIVATE cxx_std_17)
  target_include_directories(camera_lifecycle_fixture PRIVATE "${R36_PROJECT_ROOT}/features/camera/vendor/mo")
  target_link_libraries(camera_lifecycle_fixture PRIVATE Threads::Threads)
  add_executable(camera_lifecycle_test "${R36_PROJECT_ROOT}/validation/camera_lifecycle_test.cpp")
  target_compile_features(camera_lifecycle_test PRIVATE cxx_std_17)
  target_link_libraries(camera_lifecycle_test PRIVATE camera_input ${CMAKE_DL_LIBS})
  add_test(NAME camera_lifecycle COMMAND camera_lifecycle_test $<TARGET_FILE:camera_lifecycle_fixture>)
  add_executable(stereo_ros_helpers_test "${R36_PROJECT_ROOT}/validation/stereo_ros_helpers_test.cpp")
  target_compile_features(stereo_ros_helpers_test PRIVATE cxx_std_17)
  target_include_directories(stereo_ros_helpers_test PRIVATE "${R36_PROJECT_ROOT}/features/stereo/include")
  target_link_libraries(stereo_ros_helpers_test cloud_output)
  ament_target_dependencies(stereo_ros_helpers_test sensor_msgs std_msgs)
  add_test(NAME v8_stereo_ros_helpers COMMAND stereo_ros_helpers_test "${CMAKE_CURRENT_BINARY_DIR}/stereo_helper_output")
endif()
if(R36_BUILD_ALGORITHM_DEMO AND R36_BUILD_ONLINE_STEREO_TESTS)
  enable_testing()
  add_executable(stereo_provider_inputs_test "${R36_PROJECT_ROOT}/validation/stereo_provider_inputs_test.cpp")
  target_compile_features(stereo_provider_inputs_test PRIVATE cxx_std_17)
  target_include_directories(stereo_provider_inputs_test PRIVATE "${R36_PROJECT_ROOT}/compare/runner/include")
  target_include_directories(stereo_provider_inputs_test SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(stereo_provider_inputs_test PRIVATE stereo_provider)
  if(NOT R36_BUILD_HORIZON_STEREO_RUNNER)
    target_compile_definitions(stereo_provider_inputs_test PRIVATE R36_TEST_HORIZON_DISABLED=1)
  endif()
  add_test(NAME stereo_provider_inputs COMMAND stereo_provider_inputs_test)
  add_executable(latest_frame_test "${R36_PROJECT_ROOT}/validation/latest_frame_test.cpp")
  target_compile_features(latest_frame_test PRIVATE cxx_std_17)
  target_link_libraries(latest_frame_test PRIVATE r36_core_contracts Threads::Threads)
  add_test(NAME latest_frame COMMAND latest_frame_test)
