cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(raw_lux_sdk_x5_binary LANGUAGES NONE)
endif()
set(_raw_lux_root "${CMAKE_CURRENT_LIST_DIR}")
foreach(file include/percep_main/raw_lux_runner.hpp src/libraw_lux.so)
 if(NOT EXISTS "${_raw_lux_root}/${file}")
  message(FATAL_ERROR "Incomplete RAW lux binary SDK: ${_raw_lux_root}/${file}")
 endif()
endforeach()
add_library(RawLux::SDK SHARED IMPORTED GLOBAL)
set_target_properties(RawLux::SDK PROPERTIES IMPORTED_LOCATION "${_raw_lux_root}/src/libraw_lux.so" INTERFACE_INCLUDE_DIRECTORIES "${_raw_lux_root}/include" INTERFACE_COMPILE_FEATURES cxx_std_17)
install(FILES "$<TARGET_FILE:RawLux::SDK>" DESTINATION lib/ai_node)
install(DIRECTORY "${_raw_lux_root}/include/" DESTINATION include)
install(DIRECTORY "${_raw_lux_root}/config/" DESTINATION share/raw_lux_sdk_x5/config FILES_MATCHING PATTERN "*.json")
