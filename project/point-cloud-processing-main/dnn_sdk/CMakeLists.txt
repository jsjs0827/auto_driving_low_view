cmake_minimum_required(VERSION 3.16)
project(percep_dnn_parallel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PERCEP_DNN_MOCK_BACKEND "Test only: replace BPU calls with a concurrency-checking backend" OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(_standalone ON)
else()
  set(_standalone OFF)
endif()
option(PERCEP_DNN_BUILD_DEMO "Build the standalone SDK demo" ${_standalone})
set(PERCEP_DNN_INSTALL_LIBDIR "lib" CACHE STRING "DNN shared library install destination")
set(HORIZON_ROOT "" CACHE PATH "X5 runtime prefix, typically SYSROOT/usr")
find_package(Threads REQUIRED)
find_path(HORIZON_INCLUDE dnn/hb_dnn.h HINTS "${HORIZON_ROOT}/include" REQUIRED)
if(PERCEP_DNN_BUILD_DEMO OR PERCEP_DNN_MOCK_BACKEND)
  find_path(RAPIDJSON_INCLUDE rapidjson/document.h HINTS "${HORIZON_ROOT}/include" REQUIRED)
endif()
if(PERCEP_DNN_MOCK_BACKEND)
  add_library(horizon_mock SHARED tests/horizon_mock.cpp)
  target_include_directories(horizon_mock PRIVATE "${HORIZON_INCLUDE}")
  target_link_libraries(horizon_mock PRIVATE Threads::Threads)
  set(_backend horizon_mock)
else()
  find_library(HORIZON_DNN NAMES dnn HINTS "${HORIZON_ROOT}/lib" "${HORIZON_ROOT}/lib/hbbpu" REQUIRED)
  find_library(HORIZON_HBRT NAMES hbrt_bayes_aarch64 HINTS "${HORIZON_ROOT}/lib" "${HORIZON_ROOT}/lib/hbbpu" REQUIRED)
  set(_backend "${HORIZON_DNN}" "${HORIZON_HBRT}")
endif()
add_library(percep_dnn SHARED src/horizon_dnn_runner.cpp src/exact_input.cpp
  src/legacy_preprocess.cpp src/reference_preprocess.cpp src/seg_post.cpp)
target_include_directories(percep_dnn PUBLIC include PRIVATE src/internal "${HORIZON_INCLUDE}")
target_compile_options(percep_dnn PRIVATE -Wall -Wextra -fno-fast-math -ffunction-sections -fdata-sections)
# Match the production baseline's reciprocal and affine-coordinate arithmetic.
# Applying strict FP to the original scalar oracle changed thousands of NV12
# bytes by one on X5, which changed segmentation predictions.
set(_legacy_fp_options -ffast-math -funroll-loops -ffp-contract=fast)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  list(APPEND _legacy_fp_options -march=armv8.2-a)
endif()
set_source_files_properties(src/horizon_dnn_runner.cpp src/seg_post.cpp
  PROPERTIES COMPILE_OPTIONS "${_legacy_fp_options}")
set_source_files_properties(src/reference_preprocess.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=fast")
set_source_files_properties(src/exact_input.cpp src/legacy_preprocess.cpp
  PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
target_link_libraries(percep_dnn PRIVATE ${_backend} Threads::Threads rt dl)
target_link_options(percep_dnn PRIVATE -Wl,--no-undefined -Wl,--gc-sections
  "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
set_target_properties(percep_dnn PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}/runtime"
  INSTALL_RPATH "$ORIGIN;$ORIGIN/../runtime")
if(CMAKE_SYSROOT)
  target_link_options(percep_dnn PRIVATE "-Wl,-rpath-link,${CMAKE_SYSROOT}/usr/lib"
    "-Wl,-rpath-link,${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu"
    "-Wl,-rpath-link,${CMAKE_SYSROOT}/lib/aarch64-linux-gnu"
    "-Wl,-rpath-link,${CMAKE_SYSROOT}/usr/hobot/lib")
endif()
if(PERCEP_DNN_BUILD_DEMO)
add_executable(demo demo.cpp)
target_include_directories(demo PRIVATE "${RAPIDJSON_INCLUDE}")
target_link_libraries(demo PRIVATE percep_dnn)
target_link_options(demo PRIVATE "-Wl,-rpath-link,${CMAKE_CURRENT_SOURCE_DIR}/runtime")
set_target_properties(demo PROPERTIES INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN/../runtime")
if(CMAKE_SYSROOT)
  target_link_options(demo PRIVATE "-Wl,-rpath-link,${CMAKE_SYSROOT}/usr/lib"
    "-Wl,-rpath-link,${CMAKE_SYSROOT}/usr/lib/aarch64-linux-gnu"
    "-Wl,-rpath-link,${CMAKE_SYSROOT}/lib/aarch64-linux-gnu"
    "-Wl,-rpath-link,${CMAKE_SYSROOT}/usr/hobot/lib")
endif()
endif()
if(PERCEP_DNN_MOCK_BACKEND)
  enable_testing()
  add_executable(parallel_config_checks tests/parallel_config_checks.cpp)
  target_include_directories(parallel_config_checks PRIVATE src/internal "${RAPIDJSON_INCLUDE}")
  target_link_libraries(parallel_config_checks PRIVATE percep_dnn horizon_mock Threads::Threads)
  target_compile_options(parallel_config_checks PRIVATE -UNDEBUG)
  add_test(NAME parallel_config_checks COMMAND parallel_config_checks)
  add_executable(numerical_checks tests/numerical_checks.cpp src/exact_input.cpp
    src/legacy_preprocess.cpp src/reference_preprocess.cpp src/seg_post.cpp)
  target_include_directories(numerical_checks PRIVATE src/internal)
  target_compile_options(numerical_checks PRIVATE -fno-fast-math -UNDEBUG)
  add_test(NAME numerical_checks COMMAND numerical_checks)
endif()
install(TARGETS percep_dnn LIBRARY DESTINATION "${PERCEP_DNN_INSTALL_LIBDIR}")
if(PERCEP_DNN_BUILD_DEMO)
install(TARGETS demo RUNTIME DESTINATION bin)
endif()
install(DIRECTORY include/ DESTINATION include)
