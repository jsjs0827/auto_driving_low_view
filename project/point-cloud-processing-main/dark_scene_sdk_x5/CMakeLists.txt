cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(dark_scene_sdk_x5_binary LANGUAGES NONE)
endif()
set(_dark_root "${CMAKE_CURRENT_LIST_DIR}")
foreach(file include/percep_main/dark_scene_runtime.hpp src/libdark_scene.so)
 if(NOT EXISTS "${_dark_root}/${file}")
  message(FATAL_ERROR "Incomplete dark-scene SDK: ${_dark_root}/${file}")
 endif()
endforeach()
add_library(DarkScene::SDK SHARED IMPORTED GLOBAL)
set_target_properties(DarkScene::SDK PROPERTIES IMPORTED_LOCATION "${_dark_root}/src/libdark_scene.so" INTERFACE_INCLUDE_DIRECTORIES "${_dark_root}/include" INTERFACE_COMPILE_FEATURES cxx_std_17)
install(FILES "$<TARGET_FILE:DarkScene::SDK>" DESTINATION lib/ai_node)
install(FILES "${_dark_root}/include/percep_main/dark_scene_runtime.hpp" DESTINATION include/percep_main)
install(DIRECTORY "${_dark_root}/config/" DESTINATION share/dark_scene_sdk_x5/config FILES_MATCHING PATTERN "*.json")
